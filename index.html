보안 서버 설계 & process.env 활용법

왜 프론트엔드에 API 키를 노출시키면 안 되나요?

클라이언트 자바스크립트에 Gemini API Key나 NEIS API Key를 하드코딩하면 사용자의 [개발자 도구(F12)]나 소스코드에서 즉시 노출되어 무단 도용 및 과금 피해가 발생할 수 있습니다.

올바른 서버 프록시(Proxy) 구조

[브라우저/UI] ──(요청: 날짜,학교코드)──> [Express/Vercel 백엔드 서버]
                                     │ (process.env.GEMINI_API_KEY 읽기)
                                     ▼
[브라우저/UI] <──(결과: AI 한줄평)──── [Google Gemini API]

Node.js Express 백엔드 코드 (`server.js`) 예시

// server.js (Node.js 백엔드 예시)
require('dotenv').config(); // .env 파일 로드
const express = require('express');
const fetch = require('node-fetch');
const app = express();
app.use(express.json());

app.post('/api/meal-review', async (req, res) => {
  try {
    const { dishes, schoolName, date, tone } = req.body;
    
    // 서버 환경변수(process.env)에서 안전하게 Gemini API 키 참조
    const apiKey = process.env.GEMINI_API_KEY; 
    const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

    const prompt = `학교: ${schoolName}, 날짜: ${date}, 메뉴: ${dishes.join(', ')}`;
    const response = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        contents: [{ parts: [{ text: prompt }] }]
      })
    });

    const data = await response.json();
    res.json(data);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

app.listen(3000, () => console.log('Server running on port 3000'));

* 본 캔버스 시연 환경에서는 백엔드 없이도 즉시 체험하실 수 있도록 런타임 클라이언트 호출 및 키 입력 인터페이스가 통합 내장되어 있습니다. 실제 프로덕션 배포 시 위 백엔드 코드로 전환하시면 완벽합니다.
